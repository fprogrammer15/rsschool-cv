Code example

Solution of the Codewars kata Sum of Digits / Digital Root (6 kyu). The digital root is the recursive sum of all digits of a number until a single digit remains.

// Repeatedly sum the digits while the number has more than one digit
function digitalRoot(n) {
    while (n > 9) {
        n = String(n)
        .split('')
        .reduce((sum, digit) => sum + Number(digit), 0);
    }
    return n;
}

digitalRoot(16);     // 1 + 6 = 7
digitalRoot(942);    // 9 + 4 + 2 = 15, 1 + 5 = 6
digitalRoot(493193); // 4 + 9 + 3 + 1 + 9 + 3 = 29, 2 + 9 = 11, 1 + 1 = 2
